# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.19.1
engine.minor=19
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The table is also played with the pointer or touch: the Roll / Hint / Undo
# buttons, a die button (or the die on the 3D table) to hold it, a category
# on the card to score it.
control.hold0=Digit1 | Hold die 1
control.hold1=Digit2 | Hold die 2
control.hold2=Digit3 | Hold die 3
control.hold3=Digit4 | Hold die 4
control.hold4=Digit5 | Hold die 5
control.roll=KeyR | Roll
control.hint=KeyH | Hint
control.undo=KeyU | Undo turn
control.pause=Escape | Pause / close overlay
control.camera=KeyC | Camera preset
control.focus_prev=ArrowLeft | Focus previous die
control.focus_next=ArrowRight | Focus next die
control.confirm=Enter+Space+NumpadEnter | Hold focused die
engine.group.hold0=Dice
engine.group.roll=Turn
engine.group.pause=Table

# Gamepad defaults (the browser build's): the D-pad or left stick moves the
# die focus, A holds the focused die, X rolls, Y hints, B undoes the turn
# (and closes overlays), Start pauses, the right stick's press changes the
# camera.
engine.gamepad.focus_prev=DpadLeft+LeftStickLeft
engine.gamepad.focus_next=DpadRight+LeftStickRight
engine.gamepad.confirm=South
engine.gamepad.roll=West
engine.gamepad.hint=North
engine.gamepad.undo=East
engine.gamepad.pause=Start
engine.gamepad.camera=RightStickPress
